## Installation

```bash
pnpm ui:add hover-card
```

## Usage

```html
<script lang="ts">
  import * as HoverCard from "@lapismd/design-core/shadcn/hover-card";
</script>
```

```html
<HoverCard.Root>
  <HoverCard.Trigger>Hover</HoverCard.Trigger>
  <HoverCard.Content>
    SvelteKit - Web development, streamlined
  </HoverCard.Content>
</HoverCard.Root>
```
